دليل ربط الماسح بنظام الكاشير ونقاط البيع (POS)

يمكن لأي موقع كاشير أو متجر إلكتروني فتح الماسح واستقبال الأكواد الممسوحة فورياً لإضافتها لسلة المشتريات.

1 طريقة الـ JavaScript SDK (الأسهل والأسرع - 3 أسطر فقط)

موصى به

أضف ملف الـ SDK إلى صفحة الكاشير بموقعك، وافتح الماسح بنافذة منبثقة أو مضمنة واستقبل الباركود فورياً: 

<!-- 1. تضمين مكتبة SDK -->
<script src="https://asdasdzx55.github.io/urban-octo-chainsaw/js/pos-sdk.js"></script>

<!-- 2. زر فتح الماسح في شاشة الكاشير -->
<button onclick="openPosScanner()">📷 مسح باركود منتج</button>

<script>
  const posScanner = new POSScannerSDK({
    scannerUrl: 'https://asdasdzx55.github.io/urban-octo-chainsaw/',
    onScan: function(data) {
      console.log("الباركود الممسوح:", data.code);
      console.log("نوع الباركود:", data.format);
      
      // أضف المنتج مباشرة إلى فاتورة الكاشير أو سلة المشتريات:
      addProductToCart(data.code);
    }
  });

  function openPosScanner() {
    posScanner.open({ autoClose: true }); // autoClose: إغلاق النافذة تلقائياً بعد أول مسح
  }
</script>

2 طريقة الـ Iframe و PostMessage API

إذا كنت تريد تضمين الماسح داخل جزء محدد من شاشة نقطة البيع (POS Layout) والاستماع للأحداث عبر JavaScript: 

<!-- تضمين الماسح داخل الشاشة -->
<iframe src="https://asdasdzx55.github.io/urban-octo-chainsaw/?mode=pos" 
        allow="camera; microphone" 
        width="100%" height="450" style="border:none; border-radius:12px;"></iframe>

<script>
  // استقبال الباركود عند المسح مباشرة
  window.addEventListener('message', (event) => {
    if (event.data && event.data.type === 'POS_BARCODE_SCANNED') {
      const barcode = event.data.code;
      const format = event.data.format;
      
      // تنفيذ عملية البيع
      handleScannedItem(barcode);
    }
  });
</script>

3 طريقة الـ Webhook (إرسال تلقائي إلى سيرفر الكاشير)

مرر رابط الـ API الخاص بسيرفرك في عنوان URL وسيقوم الماسح بإرسال طلب POST مباشرة مع كل باركود يتم مسحه: 

// رابط تشغيل الماسح مع Webhook:
https://asdasdzx55.github.io/urban-octo-chainsaw/?mode=pos&webhook=https://api.yourpos.com/v1/scan-event

// البيانات المرسلة (JSON Payload):
{
  "type": "POS_BARCODE_SCANNED",
  "code": "6281000000000",
  "format": "EAN_13",
  "source": "camera",
  "timestamp": "2026-08-24T14:55:00.000Z"
}
